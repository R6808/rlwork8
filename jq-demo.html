jQuery 改造对照

同一功能"按座位状态筛选自习室"，左边用原生 DOM 实现，右边用 jQuery 重构，两个结果必须一致。

原生 DOM 版（改造前）

// 原生写法：逐个元素建节点、逐个按钮绑监听
var btns = document.querySelectorAll('#native-filters button');

for (var i = 0; i < btns.length; i++) {
    btns[i].addEventListener('click', function () {
        var want = this.getAttribute('data-status');
        renderNative(want);
    });
}

function renderNative(want) {
    var box = document.getElementById('native-result');
    box.innerHTML = '';                     // 清空容器

    zones.filter(function (z) {
        return want === 'all' || statusOf(z) === want;
    }).forEach(function (z) {
        var span = document.createElement('span');   // 建节点
        span.className = 'zone-chip is-' + statusOf(z);
        span.textContent = z.name + '（余' + z.free + '）';
        box.appendChild(span);                                // 插入
    });
}

jQuery 版（改造后）

// jQuery 写法：事件委托绑定一次，模板字符串一次成型
$('#jq-filters').on('click', 'button', function () {
    var want = $(this).data('status');      // 自动转类型
    $('#jq-filters button').removeClass('btn-primary')
        .addClass('btn-outline-primary');
    $(this).removeClass('btn-outline-primary')
        .addClass('btn-primary');

    var html = zones.filter(function (z) {
        return want === 'all' || statusOf(z) === want;
    }).map(function (z) {
        return '<span class="zone-chip is-' + statusOf(z) + '">'
             + z.name + '（余' + z.free + '）</span>';
    }).join('');

    $('#jq-result').html(html);                       // 一行替换内容
});

改造要点对照

对比项 原生 DOM jQuery
绑定监听 循环给每个按钮各绑一次，4 个按钮 4 个监听器；按钮若动态增加还要重新绑 on('click', 'button', fn) 事件委托，父容器绑 1 次，后加的按钮自动生效
取自定义属性 getAttribute('data-status')，取到的永远是字符串 $(this).data('status')，自动按类型转换
切换选中样式 要写 classList.remove + classList.add 两行 removeClass().addClass() 可链式调用，一行完成
生成列表 createElement + 设属性 + appendChild，一个元素三行 用 map().join('') 拼出 HTML 后 .html() 一次性写入
代码量 约 15 行 约 13 行，但省掉了循环绑定与逐节点创建
取舍 本项目里，列表动态渲染（自习室查询页）仍用原生 DOM， 因为需要给每个卡片单独绑不同参数的回调，原生写起来更直观； 筛选按钮与图表筛选改用 jQuery，事件委托省掉重复绑定，链式调用让代码更短。 也就是说：不是"全盘换 jQuery"，而是按场景挑合适的写法。
